object - player 

void EnableAIFeatures(int nFlags, int nEnable)

Scope: public

Parameters
nFlags:
aiBuildMiningBuildings = 0x10 - AI builds mine & refinery if needed
aiBuildMiningUnits = 0x20 - AI builds harvesters or container carriers if needed
aiControlMiningUnits = 0x40 - AI keeps harvesters or container carriers busy
aiBuildSupplyBuildings = 0x80 - AI builds 
aiBuildSupplyUnits = 0x100 - AI builds supply units 
aiControlSupplyUnits = 0x200 - AI assigns supply units to buildings
aiBuildPowerBuildings = 0x400 - AI takes care of energy distribution
aiBuildResearchBuildings = 0x800 - AI builds research building(s)
aiControlResearches = 0x1000 - AI decide which researches it should invent
aiControlDefense = 0x2000 - AI checks buildings periodically, and sends defensive platoons if needed
aiControlOffense = 0x4000 - AI sends periodically attack to the enemy base
aiBuildBuilders = 0x8000 - AI builds Builders if needed
aiBuildRepairers = 0x10000 - AI builds repairers and arder them tu repair base
aiBuildShadows = 0x20000 - AI builds and uses shadows
aiBuildScreamers = 0x40000- AI builds and uses screamers
aiBuildSuperAttack = 0x80000- AI builds and uses mass destruction weapon
aiEnabled = 0x100000 - AI enabled/disable
aiBuildTanks = 0x200000 - AI takes care of building ground units
aiBuildShips = 0x400000 - AI takes care of building ships
aiBuildHelicopters = 0x800000 - AI takes care of helicopters
aiBuildCivilFactory = 0x1000000 - AI builds Civil Vehicle Factory if needed
aiBuildMilitaryFactory = 0x2000000 - AI builds Military Factory if needed
aiBuildShipyard = 0x4000000 - AI builds Shipyard if needed  (only if enemy is building it too)
aiBuildBuildings = 0x7080C90 - AI builds buildings
aiBuildSpecialUnits = 0x0F8000 - AI builds screamers/schadows etc
aiRush = 0x10000000 - AI tries to make first attack as soon as possible
aiUpgradeCannons = 0x20000000 - AI upgrades cannon
aiEverything = 0xFFFFFFF2 - all flags
aiDefenseTowers = 0x40000000 - AI builds defence towers
aiRejectAlliance = 0x80000000 - AI allways reject player's offer of alliance

Return values

Description
Enable or disables AI features
